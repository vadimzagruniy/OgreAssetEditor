cmake_minimum_required(VERSION 3.21)
project(OgreAssetEditor VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(OGRE_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_COMPONENT_OVERLAY ON CACHE BOOL "" FORCE)
set(OGRE_BUILD_COMPONENT_OVERLAY_IMGUI ON CACHE BOOL "" FORCE)
set(OGRE_BUILD_COMPONENT_BITES ON CACHE BOOL "" FORCE)
set(OGRE_BUILD_RENDERSYSTEM_GL3PLUS ON CACHE BOOL "" FORCE)
set(OGRE_BUILD_RENDERSYSTEM_D3D11 ON CACHE BOOL "" FORCE)
set(OGRE_BUILD_RENDERSYSTEM_D3D9 OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_PLUGIN_STBI ON CACHE BOOL "" FORCE)
set(OGRE_BUILD_PLUGIN_ASSIMP OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_PLUGIN_BSP OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_PLUGIN_PFX OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_PLUGIN_DOT_SCENE OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_PLUGIN_PCZ OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_PLUGIN_CG OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_PLUGIN_GLSLANG OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_RENDERSYSTEM_GL OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_RENDERSYSTEM_GLES2 OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_COMPONENT_PAGING OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_COMPONENT_MESHLODGENERATOR OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_COMPONENT_TERRAIN OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_COMPONENT_VOLUME OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_COMPONENT_PROPERTY OFF CACHE BOOL "" FORCE)
set(OGRE_BUILD_COMPONENT_BULLET OFF CACHE BOOL "" FORCE)
set(OGRE_BITES_STATIC_PLUGINS ON CACHE BOOL "" FORCE)
set(OGRE_BUILD_DEPENDENCIES ON CACHE BOOL "" FORCE)
set(OGRE_DEPENDENCIES_DIR "${CMAKE_BINARY_DIR}/third_party/ogre/DependenciesComplete" CACHE PATH "" FORCE)
set(OGRE_BUILD_RENDERSYSTEM_D3D9 OFF CACHE BOOL "" FORCE)

# Ogre's Overlay CMake writes its configured imconfig.h into IMGUI_DIR. Keep
# that generated copy in the build tree and preserve the copied dependency.
set(EDITOR_IMGUI_BUILD_DIR "${CMAKE_BINARY_DIR}/third_party/imgui")
file(COPY "${CMAKE_SOURCE_DIR}/third_party/imgui/" DESTINATION "${EDITOR_IMGUI_BUILD_DIR}")
set(IMGUI_DIR "${EDITOR_IMGUI_BUILD_DIR}" CACHE PATH "Dear ImGui docking source used by Ogre Overlay" FORCE)

add_subdirectory(third_party/ogre)
add_subdirectory(src)
